import java.util.HashMap;

public class Menu {

    private String titel;
    private static boolean returned = false;
    private final HashMap<Integer, Menukeuze> menukeuzes = new HashMap<>();

    public String getTitel() {
        return this.titel;
    }

    public void setTitel(String titel) {
        this.titel = titel;
    }

    public static void setReturned(boolean returned) {
        Menu.returned = returned;
    }

    public Menu(String titel) {
        this.setTitel(titel);
    }

    private void checkIfReturned() {
        if (!returned) {
            MenuManager.getInstance().addHistory(this);
        }
        returned = false;
    }

    public void toonMenu() {

        checkIfReturned();

        System.out.println();
        System.out.printf("=== %s%n", getTitel());

        for (Integer key : this.menukeuzes.keySet()) {
            Menukeuze m = this.menukeuzes.get(key);
            m.toonMenukeuze(key);
        }

        System.out.println();
        System.out.print("Keuze: ");

        Menukeuze menukeuze = menukeuzes.get(MyScanner.nextInt());

        if (menukeuze != null) {
            menukeuze.execute();
        }
        else {
            System.out.println("ERROR: Geen Actie gedefinieerd");
        }
    }

    public void voegMenukeuzeToe(Menukeuze menukeuze) {
        menukeuzes.put(menukeuzes.size() + 1, menukeuze);
    }
}



import java.util.Set;

public class Menukeuze {

    private String titel;
    private IActie action;
    private boolean extraLegeRegel;

    public void setTitel(String titel) {
        this.titel = titel;
    }

    public void setExtraLegeRegel(boolean extraLegeRegel) {
        this.extraLegeRegel = extraLegeRegel;
    }

    public void setAction (IActie action) {
        this.action = action;
    }

    public Menukeuze(String titel, boolean extraLegeRegel, IActie action) {
        this.setTitel(titel);
        this.setExtraLegeRegel(extraLegeRegel);
        this.setAction(action);
    }

    public Menukeuze(String titel, IActie action) {
        this.setTitel(titel);
        this.setAction(action);
    }

    public void toonMenukeuze(int index) {

        if (extraLegeRegel) {
            System.out.println();
        }
        System.out.printf("%d. %s%n", index, this.titel);
    }

    public void execute() {

        if (action != null) {
            action.execute();
        }
        else {
            System.out.println("Er is geen actie gedefinieerd voor deze keuze.");
        }
    }
}



import java.util.ArrayList;
import java.util.HashMap;
import java.util.Map;

public class MenuManager {

    public static final Map<MenuType, Menu> menus = new HashMap<>();
    public final ArrayList<Menu> menuVolgorde = new ArrayList<>();

    private static MenuManager instance = null;

    public static MenuManager getInstance() {

        if (instance == null) {
            instance = new MenuManager();
        }
        return instance;
    }

    public MenuManager() {
        initialize();
    }

    public void initialize() {
        Menukeuze terug = new Menukeuze("Return", new Return());

        Menu menu = new Menu("Menu");
        menu.voegMenukeuzeToe(new Menukeuze("submenu", new Actie()));
        MenuManager.menus.put(MenuType.MENU, menu);

        Menu submenu = new Menu("subMenu");
        submenu.voegMenukeuzeToe(new Menukeuze("subsubMenu", new Actie2()));
        submenu.voegMenukeuzeToe(terug);
        MenuManager.menus.put(MenuType.SUBMENU, submenu);

        Menu subsubMenu = new Menu("subsubMenu");
        subsubMenu.voegMenukeuzeToe(terug);
        MenuManager.menus.put(MenuType.SUBSUBMENU, subsubMenu);
    }

    public void addHistory(Menu menu) {
        menuVolgorde.add(menu);
        for (Menu m : menuVolgorde) {
            System.out.println(m.getTitel());
        }
    }

    public void removeHistory() {
        menuVolgorde.remove(menuVolgorde.size() - 1);
        for (Menu m : menuVolgorde) {
            System.out.println(m.getTitel());
        }
    }

    public Menu getHistory() {
        return menuVolgorde.get(menuVolgorde.size() - 1);
    }

    public Menu getMenu(MenuType menuType) {
        return menus.get(menuType);
    }
}



public enum MenuType {
    MENU,
    SUBMENU,
    SUBSUBMENU
}




public interface IActie {

    void execute();
}




public class Return implements IActie {

    @Override
    public void execute() {
        Menu.setReturned(true);
        MenuManager.getInstance().removeHistory();
        MenuManager.getInstance().getHistory().toonMenu();
    }
}
