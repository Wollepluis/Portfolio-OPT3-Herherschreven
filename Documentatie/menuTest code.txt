import java.util.ArrayList;
import java.util.Scanner;

public class Menu {
    //Abstract class Actie extends vanuit Menukeuze
    //toonMenukeuze --> voerUit methode in Menukeuze en @Override voor verschillende acties
    //Menukeuze abstract
    private ArrayList<Menukeuze> menukeuzes = new ArrayList<>();

    public void toonMenu() {

        Scanner scanner = new Scanner(System.in);

        int counter = 0;
        for (Menukeuze mk : menukeuzes) {
            System.out.println("[" + counter + "] " + mk.naam);
            counter++;
        }


        int keuze = -1;
        boolean keuzeGemaakt = false;
        boolean firstLoop = true;

        while (!keuzeGemaakt){
            if (firstLoop){
                System.out.print("Maak een keuze: ");
            }

            try{
                keuze = new Scanner(System.in).nextInt();
            }
            catch (Exception e){

            }


            if (keuze > menukeuzes.size() - 1 ||
                    keuze < 0){
                System.out.print("Voer een geldig getal in: ");
            }
            else {
                keuzeGemaakt = true;
            }
            firstLoop = false;
        }
        menukeuzes.get(keuze).toonMenukeuze();
    }

    public void voegKeuzeToe(Menukeuze menu){
        menukeuzes.add(menu);
    }
}




public class MainMenu extends Menukeuze {

    public MainMenu(String naam) {
        super(naam);
    }

    @Override
    public void toonMenukeuze(){
        System.out.println("Dis is Mainmenu");
        menu.toonMenu();
    }
}





public class SubMenu extends Menukeuze {

    public SubMenu(String naam) {
        super(naam);
    }

    @Override
    public void toonMenukeuze(){
        System.out.println("Dit is SubMenu");
        menu.toonMenu();
    }
}





public class Menukeuze {
    public String naam;
    public Menu menu = new Menu();

    public void toonMenukeuze() {
        System.out.println(" ");
        System.out.println("Je zit in menu: " + naam);
        menu.toonMenu();
    }

    public Menukeuze(String naam){
        this.naam = naam;
    }
}






public class Main {
    public static void main(String[] args) {
        Menukeuze mainMenu = new MainMenu("Mainmenu");
        Menukeuze submenu = new Menukeuze("Submenu");

        mainMenu.menu.voegKeuzeToe(submenu);
        submenu.menu.voegKeuzeToe(mainMenu);

        mainMenu.toonMenukeuze();
    }
}
